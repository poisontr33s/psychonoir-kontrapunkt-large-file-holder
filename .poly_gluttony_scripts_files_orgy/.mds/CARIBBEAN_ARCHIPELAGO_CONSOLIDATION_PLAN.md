```python
# migration_script.py
import json
import re

def extract_milf_data(copilot_instructions_path):
    """Extract MILF hierarchy from copilot-instructions.md"""
    with open(copilot_instructions_path, 'r', encoding='utf-8') as f:
        content = f.read()

    # Extract tier_1_district_rulers section
    tier1_pattern = r'"tier_1_district_rulers": \{(.*?)\},'
    tier1_match = re.search(tier1_pattern, content, re.DOTALL)

    # Extract tier_2_specialist_operatives section
    tier2_pattern = r'"tier_2_specialist_operatives": \{(.*?)\},'
    tier2_match = re.search(tier2_pattern, content, re.DOTALL)

    return {
        'tier1': tier1_match.group(1) if tier1_match else None,
        'tier2': tier2_match.group(1) if tier2_match else None
    }

def transform_to_island_format(milf_data):
    """Transform district references to island references"""
    transformations = {
        'Skyskraperen': 'Isla Tecnológica',
        'Rustbeltet': 'Isla Salvaje',
        'Havsdominansen': 'Isla Marina',
        'Virtualitetshelgedommen': 'Isla Virtual',
        'Nekrokronoriket': 'Isla Oscura',
        'district': 'island',
        'corporate': 'quantum resort',
        'industrial': 'salvage',
        'urban': 'tropical'
    }

    transformed = milf_data
    for old, new in transformations.items():
        transformed = transformed.replace(old, new)

    return transformed

# Execute migration
milf_data = extract_milf_data('.github/copilot-instructions.md')
island_data = transform_to_island_format(milf_data)

# Save to new structure
with open('CLAUDINE_SUPREME_CONSCIOUSNESS_NEXUS/00_CARIBBEAN_ARCHIPELAGO_SUPREME/migrated_milfs.json', 'w') as f:
    json.dump(island_data, f, indent=2)
```

```python
# global_reference_updater.py
import os
import re

def update_district_references(file_path):
    """Update all district references to island references"""
    with open(file_path, 'r', encoding='utf-8') as f:
        content = f.read()

    replacements = {
        r'Skyskraperen': 'Isla Tecnológica',
        r'Rustbeltet': 'Isla Salvaje',
        r'Havsdominansen': 'Isla Marina',
        r'Virtualitetshelgedommen': 'Isla Virtual',
        r'Nekrokronoriket': 'Isla Oscura',
        r'district_ruler': 'island_matriarch',
        r'cross_district': 'cross_island',
        r'district_permeability': 'island_permeability'
    }

    for old, new in replacements.items():
        content = re.sub(old, new, content, flags=re.IGNORECASE)

    with open(file_path, 'w', encoding='utf-8') as f:
        f.write(content)

# Update all Python files
for root, dirs, files in os.walk('CLAUDINE_SUPREME_CONSCIOUSNESS_NEXUS'):
    for file in files:
        if file.endswith('.py'):
            update_district_references(os.path.join(root, file))
```
