### **TypeScript: Validate Bi-Directional Permeability**

```typescript
import emigreringData from './hierarkisk_bidireksjonell_milf_emigrering_syntetisering.json';

interface BidirectionalCorridor {
  source: string;
  target: string;
  corridor_type: string;
  entity_liaison: string;
  permeability_score: number;
}

function validatePermeability(sourceDistrict: string, targetDistrict: string): number {
  const corridors: BidirectionalCorridor[] = emigreringData.bidirectional_permeability_matrix.cross_district_corridors;
  
  const corridor = corridors.find(c => 
    c.source === sourceDistrict && c.target === targetDistrict
  );
  
  if (!corridor) {
    throw new Error(`No corridor between ${sourceDistrict} and ${targetDistrict}`);
  }
  
  return corridor.permeability_score;
}

// Example: Check SKYSKRAPEREN → RUSTBELTET permeability
const score = validatePermeability('SKYSKRAPEREN', 'RUSTBELTET');
console.log(`Permeability Score: ${score}`); // Output: 0.95 (Seamless emigrering)
```
